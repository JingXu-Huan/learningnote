-- 查询数据库原理成绩 >= 90 的学生及其成绩 （假设 在数据库中的编号是101）
select s.student_id as '学号',   -- 学生学号
       s.name       as '姓名',   -- 学生姓名
       s.major      as '专业',   -- 学生专业
       e.grade      as '成绩'    -- 成绩
from Student s
         join Enrollment e
              on s.student_id = e.student_id  -- 连接学生和选课表
where e.course_id = 101 and e.grade >= 90;  -- 只筛选成绩 >= 90 的记录
/*
查询原理：
1. JOIN 将学生表和选课表按 student_id 对应起来，生成每条选课记录对应的学生信息。
2. WHERE 筛选出成绩 >= 90 的记录。
3. SELECT 返回学生信息和成绩。
*/

-- 查询专业为 '英语' 的学生及其成绩
select s.student_id as '学号',
       s.name       as '姓名',
       s.major      as '专业',
       e.grade      as '成绩'
from Student s
         join Enrollment e
              on s.student_id = e.student_id
where major = '英语';  -- 只筛选专业为“英语”的学生

/*
查询原理：
1. JOIN 生成学生与选课信息的组合表。
2. WHERE 对学生表中的专业列进行过滤，只保留专业为“英语”的学生。
3. SELECT 返回对应学生及其成绩。
*/

-- 查询选修了全部课程（4 门）的学生姓名
select s.name AS '姓名'
from Student s
where s.student_id IN
      (select student_id
       from Enrollment
       group by student_id
       HAVING COUNT(*) = 4);  -- 选修课程数 = 4 的学生

/*
查询原理：
1. 内层子查询：
   - GROUP BY student_id 按学生分组。
   - HAVING COUNT(*)=4 统计每个学生选修课程数，筛选出正好选修 4 门的学生 ID。
2. 外层查询：
   - WHERE ... IN 使用子查询结果，选出符合条件的学生信息。
*/

-- 查询没有任何一门课程不及格（假设及格 >= 60）的学生信息
SELECT s.student_id AS 学号,
       s.name       AS 姓名,
       s.gender     AS 性别,
       s.major      AS 专业
FROM Student s
         JOIN Enrollment e ON s.student_id = e.student_id
GROUP BY s.student_id   -- 按学生分组
HAVING MIN(e.grade) >= 60;  -- 最低成绩 >= 60，说明没有不及格课程

/*
查询原理：
1. JOIN 将学生表与选课表关联，每个学生的每门课程都形成一条记录。
2. GROUP BY student_id 将记录按学生分组。
3. HAVING MIN(e.grade) >= 60 取每个学生的最低成绩，如果最低成绩 >= 60，说明该学生所有课程都及格。
*/

-- 查询选修了除了课程号 104 以外课程的学生信息
select s.student_id AS 学号,
       s.name       AS 姓名,
       s.gender     AS 性别,
       s.major      AS 专业
from Student s
where student_id in
      (select Enrollment.student_id
       from Enrollment
       where course_id != 104   -- 排除课程号 104
       group by student_id);    -- 按学生分组

/*
查询原理：
1. 内层子查询：
   - WHERE course_id != 104 筛选出选修非 104 课程的记录。
   - GROUP BY student_id 得到选修了至少一门非 104 课程的学生 ID。
2. 外层查询：
   - WHERE student_id IN 子查询结果，返回符合条件的学生信息。
