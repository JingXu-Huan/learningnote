# 8. 高级查询

## CTE（公共表表达式）

```sql
-- 基本 CTE
WITH active_users AS (
    SELECT id, username, email
    FROM users
    WHERE status = 'active'
)
SELECT * FROM active_users WHERE age > 25;

-- 递归 CTE（树形结构查询）
WITH RECURSIVE dept_tree AS (
    -- 初始查询：顶级部门
    SELECT id, name, parent_id, 1 AS level
    FROM departments
    WHERE parent_id IS NULL

    UNION ALL

    -- 递归查询：子部门
    SELECT d.id, d.name, d.parent_id, dt.level + 1
    FROM departments d
    INNER JOIN dept_tree dt ON d.parent_id = dt.id
)
SELECT * FROM dept_tree ORDER BY level, name;
```

## 窗口函数

```sql
-- 排名（与 MySQL 8.0 语法一致）
SELECT
    name, dept, salary,
    ROW_NUMBER() OVER (PARTITION BY dept ORDER BY salary DESC) AS rn,
    RANK()       OVER (PARTITION BY dept ORDER BY salary DESC) AS rk,
    DENSE_RANK() OVER (PARTITION BY dept ORDER BY salary DESC) AS drk
FROM emp;

-- LAG / LEAD（环比分析）
SELECT
    month, revenue,
    LAG(revenue, 1) OVER (ORDER BY month) AS prev_month,
    revenue - LAG(revenue, 1) OVER (ORDER BY month) AS diff
FROM monthly_report;

-- 累计求和
SELECT
    id, amount,
    SUM(amount) OVER (ORDER BY id) AS running_total
FROM orders;
```

## LATERAL JOIN（横向关联）

```sql
-- 对左表每行执行右表的关联子查询
SELECT u.username, r.latest_order_date
FROM users u
CROSS JOIN LATERAL (
    SELECT created_at AS latest_order_date
    FROM orders o
    WHERE o.user_id = u.id
    ORDER BY created_at DESC
    LIMIT 1
) r;
```

## DISTINCT ON（PostgreSQL 独有）

```sql
-- 每组取第一条（比 ROW_NUMBER 更简洁）
SELECT DISTINCT ON (dept)
    dept, name, salary
FROM emp
ORDER BY dept, salary DESC;
```

## RETURNING（增删改返回结果）

```sql
-- INSERT 返回插入的行
INSERT INTO users (username, email) VALUES ('Tom', 'tom@example.com')
RETURNING id, username, created_at;

-- UPDATE 返回修改后的行
UPDATE users SET status = 'inactive' WHERE id = 1
RETURNING *;

-- DELETE 返回被删除的行
DELETE FROM users WHERE status = 'deleted'
RETURNING id, username;
```
